  <div class="welcome-screen" id="welcome-screen">

    <button onclick="startBattle()">Iniciar Batalha</button>

  <div class="status-bar">

  </div>

  <div class="action-panel" style="display:none;">
    <button onclick="invokeUnit()">Invocar Unidade</button>
    <button onclick="useSpecial()">Habilidade Especial</button>
    <button onclick="overclock()">Overclock</button>
  </div>

  <script>
    let energy = 100;
    let unitsOnField = 0;
    const maxUnits = 4;
    const battlefield = document.getElementById("battlefield");
    let selectedUnit = null;

    const factions = ["neoblock", "daominion", "shadowhash", "satoshi"];
    const currentFaction = factions[Math.floor(Math.random() * factions.length)];

    function createGrid() {
      for (let i = 0; i < 15; i++) {
        const cell = document.createElement("div");
        cell.classList.add("cell");
        cell.dataset.index = i;
        cell.addEventListener("click", () => handleCellClick(cell));
        battlefield.appendChild(cell);
      }
    }

    function updateEnergyDisplay() {
      document.getElementById("energy").innerText = energy;
    }

    function invokeUnit() {
      if (energy < 25 || unitsOnField >= maxUnits) return;
      const emptyCells = [...document.querySelectorAll(".cell")].filter(
        cell => cell.children.length === 0
      );
      if (emptyCells.length === 0) return;

      const unit = document.createElement("div");
      unit.classList.add("unit", currentFaction);
      unit.setAttribute("data-faction", currentFaction);
      unit.addEventListener("click", (e) => {
        e.stopPropagation();
        selectUnit(unit);
      });

      const label = document.createElement("div");
      label.classList.add("faction-label");
      label.innerText = currentFaction.toUpperCase();
      unit.appendChild(label);

      const chosenCell = emptyCells[Math.floor(Math.random() * emptyCells.length)];
      chosenCell.appendChild(unit);
      energy -= 25;
      unitsOnField++;
      updateEnergyDisplay();
    }

    function useSpecial() {
      if (energy < 40) return;
      alert("Habilidade especial da facção " + currentFaction + " ativada!");
      energy -= 40;
      updateEnergyDisplay();
    }

    function overclock() {
      if (energy < 10) return;
      alert("Overclock! Unidades temporariamente reforçadas.");
      energy = 0;
      updateEnergyDisplay();
    }

    function regenerateEnergy() {
      if (energy < 100) {
        energy += 1;
        updateEnergyDisplay();
      }
    }

    function startTimer() {
      let seconds = 0;
      setInterval(() => {
        document.getElementById("timer").innerText = ++seconds;
        regenerateEnergy();
      }, 1000);
    }

    function selectUnit(unit) {
      if (selectedUnit) selectedUnit.style.border = "2px solid white";
      selectedUnit = unit;
      unit.style.border = "2px solid #00ffcc";
    }

    function handleCellClick(cell) {
      if (!selectedUnit) return;

      if (cell.children.length === 0) {
        if (energy < 5) return;
        cell.appendChild(selectedUnit);
        selectedUnit.classList.add("move-anim");
        setTimeout(() => selectedUnit.classList.remove("move-anim"), 300);
        energy -= 5;
      } else {
        const target = cell.children[0];
        const attackerFaction = selectedUnit.getAttribute("data-faction");
        const targetFaction = target.getAttribute("data-faction");

        if (target !== selectedUnit && attackerFaction !== targetFaction) {
          if (energy < 10) return;
          target.classList.add("hit-anim");
          setTimeout(() => {
            target.remove();
            selectedUnit.classList.remove("hit-anim");
          }, 400);
          energy -= 10;
        }
      }

      updateEnergyDisplay();
      selectedUnit.style.border = "2px solid white";
      selectedUnit = null;
    }

    function startBattle() {
      document.getElementById("welcome-screen").style.display = "none";
      document.querySelector(".action-panel").style.display = "flex";
      battlefield.style.display = "grid";
      createGrid();
      startTimer();
    }

    function connectWallet() {
      const walletSpan = document.getElementById("wallet");
      walletSpan.innerText = "0xABCD...1234";
    }
  </script>
