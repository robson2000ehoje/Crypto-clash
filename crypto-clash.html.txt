  <script>
    const battlefield = document.getElementById("battlefield");
    const factions = document.querySelectorAll(".faction");
    let selectedFaction = null;
    let timer;
    let timeLeft = 15;

    factions.forEach(f => {
      f.addEventListener("click", () => {
        factions.forEach(el => el.classList.remove("selected"));
        f.classList.add("selected");
        selectedFaction = f.dataset.name;
      });
    });

    function startBattle() {
      if (!selectedFaction) {
        alert("Selecione uma facção antes de iniciar!");
        return;
      }
      document.getElementById("welcome-screen").style.display = "none";
      document.querySelector(".action-panel").style.display = "flex";
      battlefield.style.display = "grid";
      createGrid();
      startTimer();
    }

    function createGrid() {
      battlefield.innerHTML = "";
      for (let i = 0; i < 15; i++) {
        const cell = document.createElement("div");
        cell.classList.add("cell");
        cell.dataset.index = i;
        cell.addEventListener("click", () => handleCellClick(cell));
        battlefield.appendChild(cell);
      }
    }

    function handleCellClick(cell) {
      if (!cell.textContent) {
        cell.textContent = selectedFaction[0]; // Representa a facção com a inicial
        cell.style.backgroundColor = "orange";
      }
    }

    function performAction(action) {
      alert(`Ação escolhida: ${action}`);
      // Aqui você pode expandir a lógica para aplicar efeitos no campo de batalha
    }

    function startTimer() {
      timeLeft = 15;
      updateTimerDisplay();
      timer = setInterval(() => {
        timeLeft--;
        updateTimerDisplay();
        if (timeLeft === 0) {
          clearInterval(timer);
          alert("Rodada encerrada! Próxima em breve...");
          startTimer();
        }
      }, 1000);
    }

    function updateTimerDisplay() {
      document.getElementById("round-timer").textContent = `Tempo restante: ${timeLeft}s`;
    }
  </script>
